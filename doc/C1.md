    Person(utilisateur, "Utilisateur", "Un employé de l'entreprise")

    Person(visiteur, "Visiteur", "Une personne pas reconnue par l'application")

    System(app, "Application", "L'application de gestion d'entreprise")

      System_Ext(apiBanque, "API Banque", "Données bancaires")
      System_Ext(serviceEmail, "Service Email", "Envoi de mails")
    }
}
